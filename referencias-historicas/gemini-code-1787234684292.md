# Contexto del Proyecto: Sistema de Inventario de Bodega y Registro de PC en Salas

## 1. Descripción General
Sistema web (MVC - Bootstrap - MySQL) para la gestión de inventario de insumos en bodega, administración de hardware (PCs, monitores, periféricos) y control de asignaciones/traslados de equipos en salas institucionales.

---

## 2. Motor de Base de Datos
* **DBMS:** MySQL 8.0+ / InnoDB
* **Base de Datos:** `inventario_institucional`
* **Codificación:** `utf8mb4_unicode_ci`

---

## 3. Script Completo de la Base de Datos (Estructura + Datos de Prueba)

```sql
-- ============================================================
-- SISTEMA DE INVENTARIO DE BODEGA Y REGISTRO DE PC EN SALAS
-- Script Completo: Creación de Esquema + Carga de Datos Iniciales
-- ============================================================

SET FOREIGN_KEY_CHECKS = 0;

DROP DATABASE IF EXISTS inventario_institucional;

CREATE DATABASE inventario_institucional
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE inventario_institucional;

-- ------------------------------------------------------------
-- 1. ESTRUCTURA DE TABLAS
-- ------------------------------------------------------------

CREATE TABLE roles (
    id_rol INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE,
    descripcion VARCHAR(150),
    estado TINYINT(1) NOT NULL DEFAULT 1
) ENGINE=InnoDB;

CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    email VARCHAR(100) UNIQUE,
    id_rol INT NOT NULL,
    estado TINYINT(1) NOT NULL DEFAULT 1,
    fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_usuario_rol
        FOREIGN KEY (id_rol) REFERENCES roles(id_rol)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

CREATE TABLE salas (
    id_sala INT AUTO_INCREMENT PRIMARY KEY,
    codigo VARCHAR(30) NOT NULL UNIQUE,
    nombre VARCHAR(100) NOT NULL,
    edificio VARCHAR(100),
    piso VARCHAR(30),
    capacidad INT,
    estado TINYINT(1) NOT NULL DEFAULT 1,
    observacion VARCHAR(255),
    fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE estados_equipo (
    id_estado_equipo INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE,
    descripcion VARCHAR(150)
) ENGINE=InnoDB;

CREATE TABLE equipos (
    id_equipo INT AUTO_INCREMENT PRIMARY KEY,
    activo_fijo VARCHAR(50) NOT NULL UNIQUE,
    codigo_barras VARCHAR(100) UNIQUE,
    tipo_equipo VARCHAR(50) NOT NULL,
    marca VARCHAR(100),
    modelo VARCHAR(100),
    numero_serie VARCHAR(100),
    id_estado_equipo INT NOT NULL,
    fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    observacion VARCHAR(255),

    CONSTRAINT fk_equipo_estado
        FOREIGN KEY (id_estado_equipo) REFERENCES estados_equipo(id_estado_equipo)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

CREATE TABLE salas_equipos (
    id_sala_equipo INT AUTO_INCREMENT PRIMARY KEY,
    id_sala INT NOT NULL,
    id_equipo INT NOT NULL,
    fecha_asignacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    id_usuario INT NOT NULL,
    observacion VARCHAR(255),

    CONSTRAINT fk_sala_equipo_sala
        FOREIGN KEY (id_sala) REFERENCES salas(id_sala)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_sala_equipo_equipo
        FOREIGN KEY (id_equipo) REFERENCES equipos(id_equipo)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_sala_equipo_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
        ON UPDATE CASCADE ON DELETE RESTRICT,

    UNIQUE (id_equipo)
) ENGINE=InnoDB;

CREATE TABLE historial_traslados (
    id_traslado INT AUTO_INCREMENT PRIMARY KEY,
    id_equipo INT NOT NULL,
    sala_origen INT NULL,
    sala_destino INT NULL,
    fecha_traslado DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    id_usuario INT NOT NULL,
    motivo VARCHAR(150),
    observacion VARCHAR(255),

    CONSTRAINT fk_traslado_equipo
        FOREIGN KEY (id_equipo) REFERENCES equipos(id_equipo)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_traslado_sala_origen
        FOREIGN KEY (sala_origen) REFERENCES salas(id_sala)
        ON UPDATE CASCADE ON DELETE SET NULL,
    CONSTRAINT fk_traslado_sala_destino
        FOREIGN KEY (sala_destino) REFERENCES salas(id_sala)
        ON UPDATE CASCADE ON DELETE SET NULL,
    CONSTRAINT fk_traslado_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

CREATE TABLE ubicaciones (
    id_ubicacion INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    tipo VARCHAR(50) NOT NULL,
    descripcion VARCHAR(150),
    estado TINYINT(1) NOT NULL DEFAULT 1,
    fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE categorias (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    descripcion VARCHAR(150),
    estado TINYINT(1) NOT NULL DEFAULT 1,
    fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE inventario_items (
    id_item INT AUTO_INCREMENT PRIMARY KEY,
    codigo_barras VARCHAR(100) UNIQUE,
    nombre VARCHAR(150) NOT NULL,
    descripcion VARCHAR(255),
    id_categoria INT NOT NULL,
    id_ubicacion INT NOT NULL,
    stock_actual INT NOT NULL DEFAULT 0,
    stock_minimo INT NOT NULL DEFAULT 0,
    costo_unitario DECIMAL(12,2) DEFAULT 0.00,
    estado TINYINT(1) NOT NULL DEFAULT 1,
    fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT fk_item_categoria
        FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_item_ubicacion
        FOREIGN KEY (id_ubicacion) REFERENCES ubicaciones(id_ubicacion)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

CREATE TABLE movimientos_inventario (
    id_movimiento INT AUTO_INCREMENT PRIMARY KEY,
    id_item INT NOT NULL,
    tipo_movimiento ENUM('Entrada', 'Salida', 'Traslado', 'Ajuste') NOT NULL,
    cantidad INT NOT NULL,
    ubicacion_origen INT NULL,
    ubicacion_destino INT NULL,
    fecha_movimiento DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    id_usuario INT NOT NULL,
    observacion VARCHAR(255),

    CONSTRAINT fk_movimiento_item
        FOREIGN KEY (id_item) REFERENCES inventario_items(id_item)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_movimiento_origen
        FOREIGN KEY (ubicacion_origen) REFERENCES ubicaciones(id_ubicacion)
        ON UPDATE CASCADE ON DELETE SET NULL,
    CONSTRAINT fk_movimiento_destino
        FOREIGN KEY (ubicacion_destino) REFERENCES ubicaciones(id_ubicacion)
        ON UPDATE CASCADE ON DELETE SET NULL,
    CONSTRAINT fk_movimiento_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
        ON UPDATE CASCADE ON DELETE RESTRICT,

    CHECK (cantidad > 0)
) ENGINE=InnoDB;

CREATE TABLE bitacora (
    id_bitacora INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    accion VARCHAR(50) NOT NULL,
    tabla_afectada VARCHAR(50),
    id_registro INT,
    detalle TEXT,
    ip VARCHAR(45),
    fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_bitacora_usuario
        FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

SET FOREIGN_KEY_CHECKS = 1;

-- ------------------------------------------------------------
-- 2. ÍNDICES DE OPTIMIZACIÓN
-- ------------------------------------------------------------
CREATE INDEX idx_equipo_activo_fijo ON equipos(activo_fijo);
CREATE INDEX idx_equipo_codigo_barras ON equipos(codigo_barras);
CREATE INDEX idx_equipo_serie ON equipos(numero_serie);
CREATE INDEX idx_item_codigo_barras ON inventario_items(codigo_barras);
CREATE INDEX idx_item_nombre ON inventario_items(nombre);
CREATE INDEX idx_movimiento_fecha ON movimientos_inventario(fecha_movimiento);
CREATE INDEX idx_historial_equipo ON historial_traslados(id_equipo);
CREATE INDEX idx_sala_equipo_sala ON salas_equipos(id_sala);
